cmake_minimum_required(VERSION 3.22)
project(Chess VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CHESS_ENABLE_COVERAGE "Instrument production code and generate informational line coverage" OFF)
option(CHESS_DEPLOY_RUNTIME "Install compiler runtime libraries on Windows" OFF)
if(CHESS_ENABLE_COVERAGE)
    include(cmake/Coverage.cmake)
endif()
find_package(Threads REQUIRED)
include(cmake/Dependencies.cmake)
include(cmake/EmbedPieces.cmake)
add_library(chess_engine ChessBackend/Position.cpp ChessBackend/ChessAI.cpp)
target_include_directories(chess_engine PUBLIC ${CMAKE_SOURCE_DIR})
add_library(chess_frontend Frontend/Game.cpp Frontend/ChessUI.cpp)
target_link_libraries(chess_frontend PUBLIC chess_engine imgui Threads::Threads)
if(CHESS_ENABLE_COVERAGE)
    target_link_libraries(chess_engine PUBLIC chess_coverage)
    target_link_libraries(chess_frontend PUBLIC chess_coverage)
endif()
if(ANDROID)
    add_library(main SHARED Frontend/main.cpp)
    set(app_target main)
else()
    add_executable(Chess Frontend/main.cpp)
    set(app_target Chess)
    # SDL main wrapper is unnecessary on desktop; keep a normal console entry point.
    target_compile_definitions(Chess PRIVATE SDL_MAIN_HANDLED)
endif()
target_include_directories(${app_target} PRIVATE ${CMAKE_BINARY_DIR})
target_link_libraries(${app_target} PRIVATE chess_frontend imgui_sdl SDL2::SDL2)
if(NOT ANDROID)
    enable_testing()
    foreach(suite position sliding_piece ai)
        add_executable(${suite}_test Tests/${suite}_test.cpp)
        target_link_libraries(${suite}_test PRIVATE chess_engine)
        add_test(NAME ${suite}_test COMMAND ${suite}_test)
        set_tests_properties(${suite}_test PROPERTIES TIMEOUT 30)
    endforeach()
    foreach(suite frontend imgui)
        add_executable(${suite}_test Tests/${suite}_test.cpp)
        target_link_libraries(${suite}_test PRIVATE chess_frontend)
        add_test(NAME ${suite}_test COMMAND ${suite}_test)
        set_tests_properties(${suite}_test PROPERTIES TIMEOUT 30)
    endforeach()
    add_test(NAME renderer_smoke COMMAND Chess --smoke-test)
    set_tests_properties(renderer_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 15)
    if(CHESS_ENABLE_COVERAGE)
        chess_add_coverage_target()
    endif()
    include(GNUInstallDirs)
    install(TARGETS Chess RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(MSVC AND CHESS_DEPLOY_RUNTIME)
        set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
        include(InstallRequiredSystemLibraries)
    endif()
    install(FILES third_party/README.md third_party/json.LICENSE third_party/stb.LICENSE
        ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION share/Chess/licenses)
    install(FILES ${sdl2_SOURCE_DIR}/LICENSE.txt DESTINATION share/Chess/licenses RENAME SDL-LICENSE.txt)
endif()
